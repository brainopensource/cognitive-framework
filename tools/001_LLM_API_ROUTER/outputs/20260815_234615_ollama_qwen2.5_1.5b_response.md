def fibonacci_memoized(n):
    """
    Compute the n-th Fibonacci number using memoization.

    Parameters:
    - n (int): The position in the Fibonacci sequence to compute.

    Returns:
    int: The n-th Fibonacci number.
    
    Raises:
    ValueError: If n is negative.
    """

    if n < 0:
        raise ValueError("Input must be a non-negative integer.")
    
    # Base cases
    if n == 0 or n == 1:
        return n

    # Memoization dictionary to store computed values
    memo = {}

    def fib(n):
        if n in memo:
            return memo[n]
        elif n > 1:
            memo[n] = fib(n - 1) + fib(n - 2)
            return memo[n]
        else:
            raise ValueError("Input must be a non-negative integer.")

    return fib(n)

# Example check function to verify the correctness
def check_fibonacci():
    assert fibonacci_memoized(0) == 0, "Test case 1 failed"
    assert fibonacci_memoized(1) == 1, "Test case 2 failed"
    assert fibonacci_memoized(5) == 5, "Test case 3 failed"
    assert fibonacci_memoized(7) == 13, "Test case 4 failed"
    print("All test cases passed!")

check_fibonacci()
